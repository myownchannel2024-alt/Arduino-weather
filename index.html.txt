<script>
function showSection(id){
  document.querySelectorAll('section').forEach(s => s.classList.remove('active'));
  document.getElementById(id).classList.add('active');
}

function getLiveData(){
  fetch('/live')
    .then(r => r.json())
    .then(data => {
      document.getElementById('temp').innerText = data.temp;
      document.getElementById('hum').innerText = data.hum;
      document.getElementById('wind').innerText = data.wind;
      document.getElementById('air').innerText = data.air;
      document.getElementById('weatherCond').innerText = data.weatherCond;
      document.getElementById('pump').innerText = data.pump;
      document.getElementById('heater').innerText = data.heater;

      document.getElementById('fTemp').innerText = data.forecast.temp;
      document.getElementById('fHum').innerText = data.forecast.hum;
      document.getElementById('fWind').innerText = data.forecast.wind;
      document.getElementById('fCond').innerText = data.forecast.condition;

      document.getElementById('sunnyAvg').innerText = data.weekly.sunny;
      document.getElementById('rainyAvg').innerText = data.weekly.rain;
      document.getElementById('cloudyAvg').innerText = data.weekly.cloudy;
      document.getElementById('windyAvg').innerText = data.weekly.windy;
      document.getElementById('snowAvg').innerText = data.weekly.snow;

      document.getElementById('sysTemp').innerText = data.temp;
      document.getElementById('sysHum').innerText = data.hum;
      document.getElementById('sysWind').innerText = data.wind;
      document.getElementById('sysRain').innerText = data.rain;
      document.getElementById('sysAir').innerText = data.air;
      document.getElementById('sysPump').innerText = data.pump;
      document.getElementById('sysHeater').innerText = data.heater;

      let tbody = document.querySelector('#forecastData tbody');
      tbody.innerHTML = '';
      data.sdLog.forEach(row => {
        let tr = document.createElement('tr');
        row.forEach(cell => {
          let td = document.createElement('td');
          td.innerText = cell;
          tr.appendChild(td);
        });
        tbody.appendChild(tr);
      });
    });
}

setInterval(getLiveData, 2000);
</script>
